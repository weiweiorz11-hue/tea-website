# Environment Variables Example

# Next.js public variables (exposed to browser)
# ------------------------------------------------
# Base path for GitHub Pages (if project name is "steelwebsite")
# NEXT_PUBLIC_BASE_PATH=/steelwebsite

# Default recipient for inquiry form (mailto fallback)
NEXT_PUBLIC_INQUIRY_TO=contact@chaoxiong-tea.com

# (Optional) API Endpoint if using a backend service
# NEXT_PUBLIC_INQUIRY_ENDPOINT=https://formspree.io/f/your-id

# ------------------------------------------------
SMTP_HOST=smtp.gmail.com
SMTP_PORT=465
SMTP_USER=contact@chaoxiong-tea.com
SMTP_PASS=your-app-password

INQUIRY_TO=contact@chaoxiong-tea.com
